title: Generate Secure Random Passwords on Linux
description: Create strong passwords using openssl rand, /dev/urandom, or pwgen commands. Multiple methods for generating cryptographically secure passwords.

  <TabsList>
    <TabsTrigger value="openssl">OpenSSL</TabsTrigger>
    <TabsTrigger value="urandom">/dev/urandom</TabsTrigger>
    <TabsTrigger value="pwgen">Password Gen</TabsTrigger>
  </TabsList>
  <TabsContent value="openssl">
    <Stepper>
      <StepperItem title="Install OpenSSL">
        ```bash
        apt install openssl
        ```
      </StepperItem>
      <StepperItem title="Generate Password">
        ```bash
        openssl rand 60 | openssl base64 -A
        ```
      </StepperItem>
    </Stepper>
  </TabsContent>
  <TabsContent value="urandom">
    ```bash
    head -c 100 /dev/urandom | LC_ALL=C tr -dc 'A-Za-z0-9!"#%&()*+,-./:;<=>?@[\]^_`{|}~' | fold -w 32 | head -n 1
    ```
  </TabsContent>
  <TabsContent value="pwgen">
    <Stepper>
      <StepperItem title="Install Pwgen">
        ```bash
        apt install pwgen
        ```
      </StepperItem>
      <StepperItem title="Generate Password">
        ```bash
        pwgen -s 64 1
        ```
      </StepperItem>
    </Stepper>
  </TabsContent>
</Tabs>
