---
title: Install Zsh and Oh My Zsh on Linux
description: Set up Zsh shell with Oh My Zsh framework on Ubuntu, Debian, CentOS, or RHEL. One-line installation command for improved terminal experience.
---

<Tabs defaultValue="debian" className="pt-5 pb-1">
  <TabsList>
    <TabsTrigger value="debian">Debian/Ubuntu</TabsTrigger>
    <TabsTrigger value="rhel">RHEL/CentOS</TabsTrigger>
  </TabsList>
  <TabsContent value="debian">
    ```bash
    apt install -y zsh git && sh -c "$(curl -fsSL https://raw.github.com/ohmyzsh/ohmyzsh/master/tools/install.sh)"
    ```
  </TabsContent>
  <TabsContent value="rhel">
    ```bash
    yum install -y zsh git && sh -c "$(curl -fsSL https://raw.github.com/ohmyzsh/ohmyzsh/master/tools/install.sh)"
    ```
  </TabsContent>
</Tabs>
