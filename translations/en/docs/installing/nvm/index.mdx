The [Node Version Manager](https://github.com/nvm-sh/nvm) is a way of managing multiple Node.js versions on one machine.
<Tabs defaultValue="debian" className="pt-5 pb-1">
  <TabsList>
    <TabsTrigger value="debian">Debian/Ubuntu</TabsTrigger>
    <TabsTrigger value="rhel">RHEL/CentOS</TabsTrigger>
    <TabsTrigger value="alpine">AlpineLinux</TabsTrigger>
  </TabsList>
  <TabsContent value="debian">
    Update your packages:
    ```bash
    apt upgrade && apt upgrade
    ```

    Run installer:
    ```bash
    curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/master/install.sh | bash
    ```
  </TabsContent>
  <TabsContent value="rhel">
    Update your packages:
    ```bash
    yum upgrade && yum upgrade
    ```

    Run installer:
    ```bash
    curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/master/install.sh | bash
    ```
  </TabsContent>
  <TabsContent value="alpine">
    Update your packages:
    ```bash
    apk upgrade --available
    ```

    Add required packages:
    ```bash
    apk add -U curl bash ca-certificates openssl ncurses coreutils python3 make gcc g++ libgcc linux-headers grep util-linux binutils findutils
    ```
    Run installer:
    ```bash
    curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/master/install.sh | bash
    ```
  </TabsContent>
</Tabs>
